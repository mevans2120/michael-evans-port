Sanity Schema: Add Footnote Annotation

Add a footnote annotation alongside the existing link annotation in sanity/schemas/blog.ts:

sanity/schemas/blog.ts (marks.annotations section) 
annotations: [
  // Existing link annotation
  {
    name: 'link',
    type: 'object',
    title: 'Link',
    fields: [
      { name: 'href', type: 'url', title: 'URL' }
    ]
  },
  // NEW: Footnote annotation
  {
    name: 'footnote',
    type: 'object',
    title: 'Footnote',
    icon: BookIcon,  // from @sanity/icons
    fields: [
      {
        name: 'note',
        type: 'text',
        title: 'Footnote Text',
        description: 'Citation or explanatory note',
        rows: 3,
        validation: (Rule) => Rule.required()
      }
    ]
  }
]

React Component: FootnoteRef

Create a simple inline component for the superscript reference:

src/components/blog/FootnoteRef.tsx 
interface FootnoteRefProps {
  number: number;
  children: React.ReactNode;
}

/**
 * FootnoteRef - Inline superscript reference
 * Wraps annotated text and adds clickable number
 */
export function FootnoteRef({ number, children }: FootnoteRefProps) {
  return (
    <>
      {children}
      <a
        href={`#fn-${number}`}
        id={`fnref-${number}`}
        className="text-accent text-[0.75em] align-super font-medium
          hover:text-accent-light hover:underline transition-smooth"
        aria-describedby={`fn-${number}`}
      >
        [{number}]
      </a>
    </>
  );
}

React Component: FootnotesSection

Renders all collected footnotes at the bottom of the article:

src/components/blog/FootnotesSection.tsx 
interface Footnote {
  _key: string;
  note: string;
}

interface FootnotesSectionProps {
  footnotes: Footnote[];
}

/**
 * FootnotesSection - Renders collected footnotes at article end
 */
export function FootnotesSection({ footnotes }: FootnotesSectionProps) {
  if (!footnotes.length) return null;

  return (
    <section
      className="mt-12 pt-6 border-t border-border"
      aria-label="Footnotes"
    >
      <h2 className="text-[11px] uppercase tracking-wide
        text-muted-foreground mb-4">
        Notes
      </h2>
      <ol className="list-none text-sm text-muted-foreground">
        {footnotes.map((fn, index) => (
          <li
            key={fn._key}
            id={`fn-${index + 1}`}
            className="mb-3 pl-6 relative scroll-mt-24
              target:bg-accent/10 target:rounded target:p-2 target:pl-8
              target:-mx-2"
          >
            <span className="absolute left-0 text-accent font-semibold">
              {index + 1}.
            </span>
            {fn.note}
            <a
              href={`#fnref-${index + 1}`}
              className="text-accent ml-2 hover:underline"
              aria-label={`Back to reference ${index + 1}`}
            >
              ↩
            </a>
          </li>
        ))}
      </ol>
    </section>
  );
}

BlogContent Integration

Update BlogContent.tsx to collect footnotes and render them:

src/components/blog/BlogContent.tsx (key changes) 
import { FootnoteRef } from './FootnoteRef';
import { FootnotesSection } from './FootnotesSection';

export function BlogContent({ content }: BlogContentProps) {
  // Collect all footnotes from markDefs across all blocks
  const footnotes = content.reduce((notes, block) => {
    if (block._type !== 'block' || !block.markDefs?.length) {
      return notes;
    }
    const blockFootnotes = block.markDefs.filter(
      (def) => def._type === 'footnote'
    );
    return [...notes, ...blockFootnotes];
  }, [] as Footnote[]);

  // Create lookup map for footnote numbers
  const footnoteNumbers = new Map(
    footnotes.map((fn, i) => [fn._key, i + 1])
  );

  const components: PortableTextComponents = {
    marks: {
      // ... existing marks ...

      // NEW: Footnote mark
      footnote: ({ value, children }) => (
        <FootnoteRef number={footnoteNumbers.get(value._key) || 0}>
          {children}
        </FootnoteRef>
      ),
    },
    // ... rest of components ...
  };

  return (
    <div>
      <PortableText value={content} components={components} />
      <FootnotesSection footnotes={footnotes} />
    </div>
  );
}

Implementation Summary

  • Schema: Add footnote annotation to blog.ts marks
  • Components: Create FootnoteRef + FootnotesSection
  • Integration: Collect footnotes in BlogContent, render at bottom
  • Styling: Uses existing design tokens (text-accent, border-border, etc.)
  • Accessibility: ARIA labels, semantic HTML, keyboard navigation
  • No JavaScript: Pure CSS scroll behavior with :target highlighting

Estimated effort: ~1-2 hours to implement and test
